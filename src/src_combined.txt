/* ---------- CDC class-specific functional descriptors ---------- */

struct cdc_header_functional_desc {
    uint8_t  bFunctionLength;     /* Size of this descriptor: 5 */
    uint8_t  bDescriptorType;     /* CS_INTERFACE = 0x24 */
    uint8_t  bDescriptorSubType;  /* Header functional descriptor = 0x00 */
    uint16_t bcdCDC;              /* CDC specification version */
} __attribute__((packed));

struct cdc_call_management_functional_desc {
    uint8_t bFunctionLength;      /* Size of this descriptor: 5 */
    uint8_t bDescriptorType;      /* CS_INTERFACE = 0x24 */
    uint8_t bDescriptorSubType;   /* Call-management descriptor = 0x01 */
    uint8_t bmCapabilities;       /* Call-management capability flags */
    uint8_t bDataInterface;       /* Interface number of CDC data interface */
} __attribute__((packed));

struct cdc_acm_functional_desc {
    uint8_t bFunctionLength;      /* Size of this descriptor: 4 */
    uint8_t bDescriptorType;      /* CS_INTERFACE = 0x24 */
    uint8_t bDescriptorSubType;   /* ACM descriptor = 0x02 */
    uint8_t bmCapabilities;       /* ACM capability flags */
} __attribute__((packed));

struct cdc_union_functional_desc {
    uint8_t bFunctionLength;      /* Size of this descriptor: 5 */
    uint8_t bDescriptorType;      /* CS_INTERFACE = 0x24 */
    uint8_t bDescriptorSubType;   /* Union descriptor = 0x06 */
    uint8_t bMasterInterface;     /* CDC communication interface number */
    uint8_t bSlaveInterface0;     /* CDC data interface number */
} __attribute__((packed));

/* combined cdc cfg descriptor  */

struct cdc_acm_configuration_desc {
    struct usb_configuration_desc configuration;

    struct usb_interface_desc     communication_iface;
    struct cdc_header_functional_desc hdr;
    struct cdc_call_management_functional_desc call_management;
    struct cdc_acm_functional_desc acm;
    struct cdc_union_functional_desc union_desc;
    struct usb_endpoint_desc      notification_endpoint;

    struct usb_interface_desc     data_iface;
    struct usb_endpoint_desc      bulk_in_ep;
    struct usb_endpoint_desc      bulk_out_ep;
} __attribute__((packed));


struct cdc_line_coding {
    uint32_t dwDTERate;           /* Bit rate in bits per second */
    uint8_t  bCharFormat;         /* 0=1 stop, 1=1.5 stop, 2=2 stop bits */
    uint8_t  bParityType;         /* 0=none, 1=odd, 2=even, etc. */
    uint8_t  bDataBits;           /* Usually 5, 6, 7, 8, or 16 */

#ifndef USB_REGS_H
#define USB_REGS_H

#include <stdint.h>

#define USB_BASE      0x40005C00UL
#define USB_PMA_BASE  0x40006000UL   // Packet memory area (shared with CAN) (512 bytes)

#define USB ((struct usb_regs_t *) USB_BASE)

struct usb_regs_t {
    volatile uint32_t EPR[8];      // 0x00-0x1C  Endpoint registers EP0R..EP7R
    volatile uint32_t RESERVED[8]; // 0x20-0x3C  Reserved
    volatile uint32_t CNTR;        // 0x40 Control register
    volatile uint32_t ISTR;        // 0x44 Interrupt status register
    volatile uint32_t FNR;         // 0x48 Frame number register
    volatile uint32_t DADDR;       // 0x4C Device address register
    volatile uint32_t BTABLE;      // 0x50 Buffer table address register
};

/*
 * Single-buffer:
 *   ADDR_TX/COUNT_TX = TX buffer
 *   ADDR_RX/COUNT_RX = RX buffer
 *
 * Double-buffer:
 *   ADDR_TX/COUNT_TX = Buffer 0
 *   ADDR_RX/COUNT_RX = Buffer 1
 */

/* Buffer Descriptor Table (BDT) 4 half-words */
struct usb_bdt_entry_t {
    volatile uint16_t ADDR_TX;   // 0x0 Transmission buffer address
    uint16_t PAD0;
    volatile uint16_t COUNT_TX;  // 0x4 Transmission byte count
    uint16_t PAD1;
    volatile uint16_t ADDR_RX;   // 0x8 Reception buffer address
    uint16_t PAD2;
    volatile uint16_t COUNT_RX;  // 0xc Reception byte count
    uint16_t PAD3;
};
/*
 * BDT assumed at PMA offset 0 (USB->BTABLE = 0).
 * If BTABLE is relocated, USB_BDT must be adjusted.
 *  the USB PMA is 512 bytes, organised as 256 half-words, 
 * but the CPU does not see it as a contiguous byte array. 
 * Each 16-bit PMA half-word is mapped to a 32-bit slot on the CPU bus 
 * The BDT (8 entries x 8 bytes) occupies PMA bytes 0..63
 */


/* BTABLE at PMA byte 0, 8 entries × 16 CPU bytes each = 128 CPU bytes.
 * Data buffers start at PMA byte 64 (CPU offset 128). */
#define USB_BD_AT(n)  ((struct usb_bdt_entry_t *)(USB_PMA_BASE + 16U * (n)))

/* 16-bit PMA access: byte offset b -> CPU addr USB_PMA_BASE + 2b */
#define PMA_BYTE(b)   (*(volatile uint16_t *)(USB_PMA_BASE + 2U * (b)))

/* PMA byte offsets of data buffers (must start >= 128 to avoid BDT overlap at 0-127) */
#define PMA_EP0_TX    64U 
#define PMA_EP0_RX    128U
#define PMA_EP1_TX    192U
#define PMA_EP1_RX    256U
#define PMA_EP2_TX    320U



/* EPnR  reg */
#define EP_ADDR_POS      0U
#define EP_STAT_TX_POS   4U
#define EP_TYPE_POS      9U
#define EP_STAT_RX_POS   12U

#define EP_DTOG_TX       (1U << 6)
#define EP_CTR_TX        (1U << 7)
#define EP_KIND          (1U << 8)
#define EP_SETUP         (1U << 11)
#define EP_DTOG_RX       (1U << 14)
#define EP_CTR_RX        (1U << 15)

#define EP_ADDR_MASK     (0xfU << EP_ADDR_POS)
#define EP_STAT_TX_MASK  (0x3U << EP_STAT_TX_POS)
#define EP_TYPE_MASK     (0x3U << EP_TYPE_POS)
#define EP_STAT_RX_MASK  (0x3U << EP_STAT_RX_POS)

/* endpoint address encoding */
#define EP_ADDR(x)       ((x) << EP_ADDR_POS)

/* endpoint status encoding */
#define EP_TX_STAT(x)    ((x) << EP_STAT_TX_POS)
#define EP_RX_STAT(x)    ((x) << EP_STAT_RX_POS)

/* endpoint types */
enum {
    EP_TYPE_BULK    = (0U << EP_TYPE_POS),
    EP_TYPE_CONTROL = (1U << EP_TYPE_POS),
    EP_TYPE_ISO     = (2U << EP_TYPE_POS),
    EP_TYPE_INT     = (3U << EP_TYPE_POS)
};

/* Endpoint status */
enum {
    EP_DISABLED = 0,
    EP_STALL,
    EP_NAK,
    EP_VALID
};  

/* ---------- BDT COUNT_RX ---------- */

#define COUNT_NBYTES_POS         0U
#define COUNT_NBYTES_MASK        0x03FFU

#define COUNT_RX_NUM_BLOCK_POS   10U
#define COUNT_RX_NUM_BLOCK_MASK  (0x1FU << COUNT_RX_NUM_BLOCK_POS)

#define COUNT_RX_BLSIZE_POS      15U
#define COUNT_RX_BLSIZE_MASK     (1U << COUNT_RX_BLSIZE_POS)

/* 64-byte RX buffer */
#define COUNT_RX_64B ((1U << COUNT_RX_BLSIZE_POS) | (1U << COUNT_RX_NUM_BLOCK_POS))


/* ISTR bits */
#define ISTR_EP_ID      (0xfU << 0)
#define ISTR_ESOF       (0x1U << 8)
#define ISTR_SOF        (0x1U << 9)
#define ISTR_RESET      (0x1U << 10)
#define ISTR_SUSP       (0x1U << 11)
#define ISTR_WKUP       (0x1U << 12)
#define ISTR_CTR        (0x1U << 15) /* Correct transfer */

/* CNTR bits */
#define CNTR_FRES       (0x1U << 0) 
#define CNTR_PDWN       (0x1U << 1) 
#define CNTR_LPMODE     (0x1U << 2) 
#define CNTR_FSUSP      (0x1U << 3) 
#define CNTR_RESUME     (0x1U << 4) 
#define CNTR_RESETM     (0x1U << 10)
#define CNTR_SUSPM      (0x1U << 11)
#define CNTR_WKUPM      (0x1U << 12)
#define CNTR_ERRM       (0x1U << 13)
#define CNTR_PMAOVRM    (0x1U << 14)
#define CNTR_CTRM       (0x1U << 15)

#define DADDR_EF        (0x1U << 7) /* en function */

#endif /* USB_REGS_H */

#include <usb-core.h>

/*
 * CDC requires 2 interfaces, 3 endpoints, and complex union/call-management descriptors
 * per modemax bulk packet sz
 * Low-Speed  1.5 Mbit/s 8 bytes
 * Full-Speed 12 Mbit/s 64 bytes
 * High-Speed 480 Mbit/s 512 bytes
*/


static uint8_t    usb_addr;
static enum usb_ctrl_state_t  ctrl_pending;
static uint8_t    ctrl_buf[128]; /* internal buffer used for control transfers */
static uint32_t   ctrl_len;
static uint32_t   ctrl_pos;

// static uint8_t  bulk_tx_busy;
uint8_t  bulk_tx_busy;
static uint8_t  bulk_tx_buf[64];

void usb_clock_init(void) {
    /* Flash: 2 wait states + prefetch (needed for 72 MHz on F103) */
    flash_set_latency(FREQ_72MHZ);

    RCC->CR |= (1U << 16);                       /* HSEON  */
    while (!(RCC->CR & (1U << 17))) { }           /* HSERDY */

    RCC->CFGR = ((uint32_t)CLOCK_PLL   << 0)
              | ((uint32_t)AHB_DIV_1   << 4)
              | ((uint32_t)APB_DIV_2   << 8)
              | ((uint32_t)APB_DIV_1   << 11)
              | (1U << 16)                       /* PLLSRC = HSE */
              | ((uint32_t)PLL_MUL_9   << 18)
              | ((uint32_t)USB_DIV_1_5 << 22);

    RCC->CR |= (1U << 24);                       /* PLLON  */
    while (!(RCC->CR & (1U << 25))) { }           /* PLLRDY */

    RCC->CFGR = (RCC->CFGR & ~0x3U) | CLOCK_PLL; /* switch SYSCLK */
    while (((RCC->CFGR >> 2) & 0x3U) != (uint32_t)CLOCK_PLL) { }
}

/* ---------- PMA helpers (16-bit only, 2:1 hole) ---------- */
static void pma_write16(uint32_t b, const uint8_t *s, uint32_t n) {
    uint32_t i;
    for (i = 0; i < n; i += 2) {
        uint16_t w = s[i];
        if (i + 1 < n) w |= (uint16_t)s[i + 1] << 8;
        PMA_BYTE(b + i) = w;
    }
}
static void pma_read16(uint32_t b, uint8_t *d, uint32_t n) {
    uint32_t i;
    for (i = 0; i < n; i += 2) {
        uint16_t w = PMA_BYTE(b + i);
        d[i] = (uint8_t)(w & 0xFF);
        if (i + 1 < n) d[i + 1] = (uint8_t)(w >> 8);
    }
}

/* ---------- EPnR and packet transfer events handlers --------- */
static inline void ep_set(uint32_t ep, uint32_t mask, uint32_t data)
{
    uint32_t old = USB->EPR[ep];
    uint32_t rc_w0 = EP_CTR_RX | EP_CTR_TX;
    uint32_t toggle = EP_DTOG_RX | EP_STAT_RX_MASK | EP_DTOG_TX | EP_STAT_TX_MASK;
    uint32_t rw = EP_TYPE_MASK | EP_KIND | EP_ADDR_MASK;

    uint32_t wr0 = rc_w0 & (~mask | data);
    uint32_t wr1 = (mask & toggle) & (old ^ data);
    uint32_t wr2 = rw & ((old & ~mask) | data);

    USB->EPR[ep] = (uint16_t)(wr0 | wr1 | wr2);
}

static void ep_set_tx(uint32_t ep, uint32_t stat)
{
    ep_set(ep, EP_STAT_TX_MASK, EP_TX_STAT(stat));
}
static void ep_set_rx(uint32_t ep, uint32_t stat)
{
    ep_set(ep, EP_STAT_RX_MASK, EP_RX_STAT(stat));
}
static void ep_clear_ctr_rx(uint32_t ep)
{
    ep_set(ep, EP_CTR_RX, 0);
}
static void ep_clear_ctr_tx(uint32_t ep)
{
    ep_set(ep, EP_CTR_TX, 0);
}

/* ---------- init ---------- */
void usb_init(void) {
    uint32_t d;

    RCC->APB2ENR |= (1U << 2);  /* GPIOA-EN */
    RCC->APB1ENR |= (1U << 23); /* USB-EN  */

    /* PA11 / PA12 as input floating */
    GPIOA->CRH = (GPIOA->CRH & ~0x000FF000U) | 0x00044000U;
    delay_ms(10);

    USB->CNTR = CNTR_FRES;
    delay_ms(10);

    USB->CNTR = 0;
    USB->ISTR = 0;
    USB->BTABLE = 0; /* BTABLE at PMA byte 0 */

    /* EP0: control, RX valid, TX NAK */
    ep_set(0, 0xFFFFU, EP_TYPE_CONTROL | EP_TX_STAT(EP_NAK) | EP_RX_STAT(EP_VALID));
    USB_BD_AT(0)->ADDR_RX = PMA_EP0_RX; USB_BD_AT(0)->COUNT_RX = COUNT_RX_64B;
    USB_BD_AT(0)->ADDR_TX = PMA_EP0_TX; USB_BD_AT(0)->COUNT_TX = 0;

    USB->CNTR  = CNTR_CTRM | CNTR_RESETM | CNTR_SUSPM | CNTR_WKUPM;
    NVIC->ISER[0] = (1U << 20);   /* NVIC USB_LP */
}

/* ---------- control-transfer helpers ---------- */
static void ep0_send_chunk(void) {
    uint32_t n = ctrl_len - ctrl_pos;
    if (n > 64) n = 64;
    pma_write16(PMA_EP0_TX, ctrl_buf + ctrl_pos, n);
    USB_BD_AT(0)->COUNT_TX = (uint16_t)n;
    ctrl_pos += n;
    ctrl_pending = CTRL_DATA_IN;
    ep_set_tx(0, EP_VALID);
}

static void ep0_setup(void) {
    uint8_t s[8];
    uint32_t bmReq, bReq, wVal, wLen;

    pma_read16(PMA_EP0_RX, s, 8);
    ep_clear_ctr_rx(0);
    bmReq = s[0];
    bReq = s[1];
    wVal  = s[2] | ((uint32_t)s[3] << 8);
    wLen  = s[6] | ((uint32_t)s[7] << 8);

    ctrl_len = 0; ctrl_pos = 0; ctrl_pending = CTRL_IDLE;

    if ((bmReq & 0x60U) == USB_REQ_STANDARD) {
        if (bReq == REQ_GET_DESCRIPTOR) {
            uint32_t type = wVal >> 8;
            const uint8_t *src = 0; uint32_t n = 0;
            if      (type == USB_DESC_DEVICE) { src = devd; n = sizeof(devd); }
            else if (type == USB_DESC_CONFIGURATION) { src = cfgd; n = sizeof(cfgd); }
            else if (type == USB_DESC_STRING && (wVal & 0xFFU) == 0) { src = str0; n = sizeof(str0);}
            if (!src) { ep_set_tx(0, EP_STALL); ep_set_rx(0, EP_STALL); return; }
            if (n > sizeof(ctrl_buf)) n = sizeof(ctrl_buf);
            if (n > wLen) n = wLen;
            { uint32_t i; for (i = 0; i < n; i++) ctrl_buf[i] = src[i]; }
            ctrl_len = n;
            ep0_send_chunk();
        } else if (bReq == REQ_SET_ADDRESS) {
            usb_addr = (uint8_t)(wVal & 0x7FU);
            USB_BD_AT(0)->COUNT_TX = 0;
            ctrl_pending = CTRL_SET_ADDRESS;
            ep_set_tx(0, EP_VALID);
        } else if (bReq == REQ_SET_CONFIGURATION) {
            if (wVal) {
                USB_BD_AT(1)->ADDR_TX = PMA_EP1_TX;  USB_BD_AT(1)->COUNT_TX = 0;
                USB_BD_AT(1)->ADDR_RX = PMA_EP1_RX;  USB_BD_AT(1)->COUNT_RX = COUNT_RX_64B;
                ep_set(1, 0xFFFFU, EP_ADDR(1) | EP_TYPE_BULK | EP_TX_STAT(EP_NAK) | EP_RX_STAT(EP_VALID));
                USB_BD_AT(2)->ADDR_TX = PMA_EP2_TX;  USB_BD_AT(2)->COUNT_TX = 0;
                ep_set(2, 0xFFFFU, EP_ADDR(2) | EP_TYPE_INT | EP_TX_STAT(EP_NAK) | EP_RX_STAT(EP_DISABLED));
            }
            USB_BD_AT(0)->COUNT_TX = 0;
            ctrl_pending = CTRL_STATUS_IN;
            ep_set_tx(0, EP_VALID);
        } else if (bReq == REQ_GET_STATUS) {
            ctrl_buf[0] = 0; ctrl_buf[1] = 0;
            ctrl_len = 2;
            ep0_send_chunk();
        } else if (bReq == REQ_CLEAR_FEATURE || bReq == REQ_SET_FEATURE) {
            USB_BD_AT(0)->COUNT_TX = 0;
            ctrl_pending = CTRL_STATUS_IN;
            ep_set_tx(0, EP_VALID);
        } else if (bReq == REQ_SET_INTERFACE) {
            USB_BD_AT(0)->COUNT_TX = 0;
            ctrl_pending = CTRL_STATUS_IN;
            ep_set_tx(0, EP_VALID);
        } else if (bReq == REQ_GET_INTERFACE) {
            ctrl_buf[0] = 0;
            ctrl_len = 1;
            ep0_send_chunk();
        } else {
            ep_set_tx(0, EP_STALL); ep_set_rx(0, EP_STALL);
        }
    } else if ((bmReq & 0x60U) == USB_REQ_CLASS) {
        if (bReq == CDC_SET_LINE_CODING) {
            ctrl_len = (wLen < 7) ? wLen : 7;
            ctrl_pos = 0;
            ctrl_pending = CTRL_DATA_OUT;
            ep_set_rx(0, EP_VALID);
        } else if (bReq == CDC_GET_LINE_CODING) {
            uint32_t i; for (i = 0; i < 7; i++) ctrl_buf[i] = lc[i];
            ctrl_len = 7; ep0_send_chunk();
        } else if (bReq == CDC_SET_CONTROL_LINE_STATE) {
            USB_BD_AT(0)->COUNT_TX = 0;
            ctrl_pending = CTRL_STATUS_IN;
            ep_set_tx(0, EP_VALID);
        } else {
            ep_set_tx(0, EP_STALL); ep_set_rx(0, EP_STALL);
        }
    } else {
        ep_set_tx(0, EP_STALL); ep_set_rx(0, EP_STALL);
    }
}

static void ep0_in(void) {
    ep_clear_ctr_tx(0);
    if (ctrl_pending == CTRL_SET_ADDRESS) {
        USB->DADDR = usb_addr | DADDR_EF;
        ctrl_pending = CTRL_IDLE;
        ep_set_rx(0, EP_VALID);
    } else if (ctrl_pending == CTRL_DATA_IN) {
        if (ctrl_pos < ctrl_len) { ep0_send_chunk(); return; }
        ctrl_pending = CTRL_STATUS_OUT;
        ep_set_rx(0, EP_VALID);
    } else if (ctrl_pending == CTRL_STATUS_IN) {
        ctrl_pending = CTRL_IDLE;
        ep_set_rx(0, EP_VALID);
    }
}

static void ep0_out(void) {
    uint32_t cnt = USB_BD_AT(0)->COUNT_RX & COUNT_NBYTES_MASK;
    ep_clear_ctr_rx(0);
    if (ctrl_pending == CTRL_DATA_OUT) {
        pma_read16(PMA_EP0_RX, ctrl_buf + ctrl_pos, cnt);
        ctrl_pos += cnt;
        if (ctrl_pos >= ctrl_len) {
            uint32_t i; for (i = 0; i < 7; i++) lc[i] = ctrl_buf[i];
            USB_BD_AT(0)->COUNT_TX = 0;
            ctrl_pending = CTRL_STATUS_IN;
            ep_set_tx(0, EP_VALID);
        } else {
            ep_set_rx(0, EP_VALID);
        }
    } else if (ctrl_pending == CTRL_STATUS_OUT) {
        ctrl_pending = CTRL_IDLE;
        ep_set_rx(0, EP_VALID);
    } else {
        ep_set_rx(0, EP_VALID);
    }
}

/* ---------- Bulk IN driver API (fixed flow control) ---------- */
int cdc_send_data(const uint8_t *data, uint32_t len) {
    if (bulk_tx_busy || len == 0 || len > 64) return -1;
    { uint32_t i; for (i = 0; i < len; i++) bulk_tx_buf[i] = data[i]; }
    bulk_tx_len  = len;
    bulk_tx_busy = 1;
    pma_write16(PMA_EP1_TX, bulk_tx_buf, len);
    USB_BD_AT(1)->COUNT_TX = (uint16_t)len;
    ep_set_tx(1, EP_VALID);
    return 0;
}

static void ep1_out(void) {
    uint8_t buf[64];
    uint32_t cnt = USB_BD_AT(1)->COUNT_RX & COUNT_NBYTES_MASK;
    ep_clear_ctr_rx(1);
    pma_read16(PMA_EP1_RX, buf, cnt);

    if (cnt > 0) {
        if (buf[0] == 1) {
            GPIOC->BRR = (1U << 13);   /* LED ON (pull low) */
        } else if (buf[0] == 0) {
            GPIOC->BSRR = (1U << 13);  /* LED OFF (pull high) */
            }
        /* echo back, but only if the IN path is free */
        // if (!bulk_tx_busy && cnt > 0) {
        //     cdc_send_data(buf, cnt);
        // }
    }
    ep_set_rx(1, EP_VALID);
}

/* ---------- ISR ---------- */
void USB_LP_CAN1_RX0_IRQHandler(void) {
    uint32_t istr = USB->ISTR;

    if (istr & ISTR_RESET) {
        USB->ISTR = ~ISTR_RESET;
        usb_addr = 0;
        ctrl_pending = CTRL_IDLE;
        bulk_tx_busy = 0;
        USB->DADDR = DADDR_EF;  /* EF, ADD[0-6] = 0 */
        USB->CNTR  = 0;
        ep_set(0, 0xFFFFU, EP_TYPE_CONTROL | EP_TX_STAT(EP_NAK) | EP_RX_STAT(EP_VALID));
        USB_BD_AT(0)->ADDR_RX = PMA_EP0_RX;
        USB_BD_AT(0)->COUNT_RX = COUNT_RX_64B;  // 64 byte == BLSIZE=1 -> 32B units, NUM_BLOCK=1 -> 2 blocks
        USB_BD_AT(0)->ADDR_TX = PMA_EP0_TX;
        USB_BD_AT(0)->COUNT_TX = 0;

        USB->CNTR = CNTR_CTRM | CNTR_RESETM | CNTR_SUSPM | CNTR_WKUPM;
        return;
    }
    if (istr & ISTR_WKUP) {
        USB->ISTR = ~ISTR_WKUP;
        return;
    }
    if (istr & ISTR_SUSP) {
        USB->ISTR = ~ISTR_SUSP;
        return;
    }
    if (istr & ISTR_CTR) {
        uint32_t ep = (istr & ISTR_EP_ID);
        uint32_t r  = USB->EPR[ep];
        if (ep == 0) {
            if      (r & EP_SETUP)  ep0_setup();
            else if (r & EP_CTR_TX) ep0_in();
            else if (r & EP_CTR_RX) ep0_out();
        } else if (ep == 1) {
            if      (r & EP_CTR_RX) ep1_out();
            else if (r & EP_CTR_TX) {
                ep_clear_ctr_tx(1);
                bulk_tx_busy = 0;               /* IN complete, safe to reuse */
            }
        } else {
            USB->EPR[ep] = EP_CTR_TX | EP_CTR_RX;
        }
    }
    /* Clear any other unhandled ISTR bits like ESOF (0x0100) */
    if (istr & (ISTR_SOF | ISTR_ESOF)) {
        USB->ISTR = ~(istr & (ISTR_SOF | ISTR_ESOF));
    }
}

#ifndef GT911_H
#define GT911_H

#include <stdint.h>
#include <i2c.h>

#define GT911_ADDR_14               0x14U
#define GT911_ADDR_5D               0x5DU

#define GT911_REG_PRODUCT_ID        0x8140U
#define GT911_REG_FIRMWARE_VERSION  0x8144U
#define GT911_REG_X_RESOLUTION      0x8146U
#define GT911_REG_Y_RESOLUTION      0x8148U
#define GT911_REG_POINT_STATUS      0x814EU
#define GT911_REG_POINT_DATA        0x814FU

#define GT911_MAX_TOUCHES           5U

struct gt911_point_t {
    uint8_t id;
    uint16_t x;
    uint16_t y;
    uint16_t size;
};

/* gt911 expects two's complement additive checksum (~sum) + 1U <=> (0U - sum) */
static uint8_t calc_checksum(const uint8_t *cfg, uint16_t len)
{
    uint8_t sum = 0U;
    for (uint16_t i = 0U; i < len; i++) {
        sum = (uint8_t)(sum + cfg[i]);
    }
    return (uint8_t)(0U - sum);
}

extern volatile uint8_t gt911_irq_flag;

int gt911_init(volatile struct i2c_t *i2c);
int gt911_read_points(struct gt911_point_t *points, uint8_t *count);

#endif /* GT911_H */

===== ./gt911/gt911.c =====
#include <gt911.h>
#include <gpio.h>
#include <rcc.h>
#include <systick.h>
#include <alter-func.h>
#include <external-intr.h>
#include <nested-vector.h>

/* 
 GT911 requires 16-bit big-endian reg addr, we pack that addr in raw i2c read/write 
 rst sequence : pulling (PB8) low for 15 ms while INT (PB5) low latches 0x5D addr
 INT pin must be floating input after reset sequence and trigger interrupt based on gt911 0x804D reg.
  */

volatile uint8_t gt911_irq_flag = 0;

static int gt911_read_reg(volatile struct i2c_t *i2c, uint16_t reg, uint8_t *buf, uint16_t len)
{
    uint8_t addr_bytes[2] = { (uint8_t)(reg >> 8), (uint8_t)(reg & 0xFFU) };

    if (i2c_write(i2c, GT911_ADDR_5D, addr_bytes, 2U) != I2C_OK) return -1;
    return i2c_read(i2c, GT911_ADDR_5D, buf, len);
}

static int gt911_write_reg(volatile struct i2c_t *i2c, uint16_t reg, const uint8_t *data, uint16_t len)
{
    uint8_t tx[256];
    if (len > sizeof(tx) - 2U) return -1;

    tx[0] = (uint8_t)(reg >> 8);
    tx[1] = (uint8_t)(reg & 0xFFU);
    for (uint16_t i = 0; i < len; i++) tx[2U + i] = data[i];

    return i2c_write(i2c, GT911_ADDR_5D, tx, (uint16_t)(len + 2U));
}

int gt911_init(volatile struct i2c_t *i2c)
{
    rcc_gpiob_enable();
    rcc_afio_enable();

    /* 1. Reset sequence: PB8 (RST) + PB5 (INT) LOW -> RST HIGH -> INT floating */
    gpio_pin_mode(GPIOB, PIN_8, GPIO_OUTPUT_PP_2MHZ);
    gpio_pin_mode(GPIOB, PIN_5, GPIO_OUTPUT_PP_2MHZ);
    GPIOB->BRR = (1U << 8) | (1U << 5);
    delay_ms(15);

    GPIOB->BSRR = (1U << 8);
    delay_ms(6);

    gpio_pin_mode(GPIOB, PIN_5, GPIO_INPUT_FLOATING);
    delay_ms(60);

    /* 2. Verify Product ID ("911"), read INT polarity (0x804D), clear status */
    uint8_t id[4] = {0}, mod_sw1 = 0, zero = 0;
    if (gt911_read_reg(i2c, GT911_REG_PRODUCT_ID, id, 4U) != I2C_OK) return -1;
    if (id[0] != '9' || id[1] != '1' || id[2] != '1') return -1;
    if (gt911_read_reg(i2c, 0x804DU, &mod_sw1, 1U) != I2C_OK) return -1;
    if (gt911_write_reg(i2c, GT911_REG_POINT_STATUS, &zero, 1U) != I2C_OK) return -1;

    /* 3. Route PB5 to EXTI5 matching panel INT edge (0 = rising, 1 = falling) */
    AFIO->EXTICR2 = (AFIO->EXTICR2 & ~(0xFU << 4)) | (0x1U << 4);
    if ((mod_sw1 & 0x03U) == 1U) {
        EXTI->FTSR |=  (1U << 5);
        EXTI->RTSR &= ~(1U << 5);
    } else {
        EXTI->RTSR |=  (1U << 5);
        EXTI->FTSR &= ~(1U << 5);
    }

    EXTI->PR   = (1U << 5);
    EXTI->IMR |= (1U << 5);
    gt911_irq_flag = 0;
    NVIC_EnableIRQ(EXTI9_5_IRQn);

    return 0;
}

int gt911_read_points(struct gt911_point_t *points, uint8_t *count)
{
    EXTI->IMR &= ~(1U << 5); // Mute PB6 (SCL) crosstalk during I2C transfer (mask)

    uint8_t status = 0, zero = 0;
    *count = 0;

    if (gt911_read_reg(I2C1, GT911_REG_POINT_STATUS, &status, 1U) == I2C_OK && (status & 0x80U)) {
        uint8_t n = status & 0x0FU;
        if (n > GT911_MAX_TOUCHES) n = GT911_MAX_TOUCHES;

        if (n > 0U) {
            uint8_t raw[GT911_MAX_TOUCHES * 8U];
            if (gt911_read_reg(I2C1, GT911_REG_POINT_DATA, raw, (uint16_t)(n * 8U)) == I2C_OK) {
                for (uint8_t i = 0; i < n; i++) {
                    uint8_t *p = &raw[i * 8U];
                    points[i].id   = p[0];
                    points[i].x    = (uint16_t)p[1] | ((uint16_t)p[2] << 8);
                    points[i].y    = (uint16_t)p[3] | ((uint16_t)p[4] << 8);
                    points[i].size = (uint16_t)p[5] | ((uint16_t)p[6] << 8);
                }
                *count = n;
            }
        }
        gt911_write_reg(I2C1, GT911_REG_POINT_STATUS, &zero, 1U);
    }

    EXTI->PR   = (1U << 5);
    EXTI->IMR |= (1U << 5);
    return 0;
}

void crt0_init(void){
    extern volatile uint8_t  _sbss[], _ebss [], _sidata[];
    extern volatile uint8_t  _sdata[], _edata[];
    extern void (* const g_pfnVectors[])(void);

    volatile uint32_t *src, *dst;
    volatile uint8_t *byte_src, *byte_dst;

    SCB->VTOR = (uint32_t)g_pfnVectors;

    // Copy initialized data from Flash to RAM (optimize for large chunks)
    src = (volatile uint32_t *)&_sidata;
    dst = (volatile uint32_t *) &_sdata;
    while (dst + 1 <= (volatile uint32_t *) &_edata) {  *dst++ = *src++;  }
    // Handle remaining bytes (if the section size isn't a multiple of word size)
    byte_src = (volatile uint8_t *) src;
    byte_dst = (volatile uint8_t *) dst;
    while (byte_dst < (volatile uint8_t *) &_edata) { *byte_dst++ = *byte_src++;  }
    // Clear the .bss section (optimize for large chunks)
    dst = (volatile uint32_t *) &_sbss;
    while (dst + 1 <= (volatile uint32_t *) &_ebss) { *dst++ = 0; }
    // Handle remaining bytes in .bss
    byte_dst = (volatile uint8_t *) dst;
    while (byte_dst < (volatile uint8_t *) &_ebss) { *byte_dst++ = 0; }
}

void *memset(void *dst, int c, size_t n)
{
    uint8_t *p = (uint8_t *)dst;
    while (n--) {
        *p++ = (uint8_t)c;
    }
    return dst;
}

uint32_t irq_count = 0; // debug

extern uint8_t  bulk_tx_busy;

/* tiny unsigned->decimal-ASCII, no libc dependency */
static unsigned u16_to_dec(unsigned val, char *out)
{
    char tmp[6];
    unsigned n = 0;
    if (val == 0U) { out[0] = '0'; return 1U; }
    while (val > 0U && n < sizeof(tmp)) {
        tmp[n++] = (char)('0' + (val % 10U));
        val /= 10U;
    }
    for (unsigned i = 0; i < n; i++) out[i] = tmp[n - 1U - i];
    return n;
}

/* Builds "id,x,y;id,x,y;...\n" for up to GT911_MAX_POINTS points.
   Bails out early if it would overflow max_len (keeps us under 64B/packet). */
static unsigned format_multi_touch(char *line, unsigned max_len, const struct gt911_point_t *pts, unsigned n)
{
    unsigned p = 0;

    if (max_len == 0)
        return 0;

    for (unsigned i = 0; i < n; i++) {

        unsigned id = pts[i].id;
        unsigned x  = pts[i].x;
        unsigned y  = pts[i].y;

        /* Reserve space for newline */
        if (p + 14U >= max_len)
            break;

        p += u16_to_dec(id, line + p);
        line[p++] = ',';

        p += u16_to_dec(x, line + p);
        line[p++] = ',';

        p += u16_to_dec(y, line + p);
        line[p++] = ';';
    }

    line[p++] = '\n';

    return p;
}


int main(void)
{
    /* PC13 LED */
    RCC->APB2ENR |= (1U << 4);

    GPIOC->CRH = (GPIOC->CRH & ~0x00F00000U) | 0x00200000U;

    GPIOC->BSRR = (1U << 13);

    /* USB clock: 72 MHz SYSCLK, 48 MHz USB */
    usb_clock_init();

    /* I2C1 */
    i2c_init(I2C1, FREQ_400KHZ);

    /* GT911 initialization */
    if (gt911_init(I2C1) != 0) {
        while (1);
    }

    /* Enable interrupts */
    __asm__ volatile ("cpsie i");

    /* USB device initialization */
    usb_init();

    struct gt911_point_t points[GT911_MAX_TOUCHES] = {0};
    uint8_t count = 0;
    char line[64];
    unsigned len;



    while (1)
    {
        if (gt911_irq_flag)
        {
            gt911_irq_flag = 0;
            if (gt911_read_points(points, &count) == 0)
            {
                if (count > 0)
                {
                    len = format_multi_touch(line, sizeof(line), points, count);
                    if (!bulk_tx_busy && len > 0 && len <= 64)
                    {
                        cdc_send_data((const uint8_t *)line, len);
                    }
                }
            }
        }

        /* Wait for touch interrupt or USB interrupt */
        __asm__ volatile ("wfi");
    }

    return 0;
}
